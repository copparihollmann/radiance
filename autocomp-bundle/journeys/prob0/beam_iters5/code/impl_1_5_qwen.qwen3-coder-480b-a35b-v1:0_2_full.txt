#include <cstdint>

// Load a float from shared memory with a compile-time immediate offset
inline float load32_shared_offset(uint32_t address, uint32_t byte_offset) {
  uint32_t raw;
  asm volatile("lw.shared %0, %1(%2)" : "=r"(raw) : "I"(byte_offset), "r"(address) : "memory");
  return __builtin_bit_cast(float, raw);
}

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* args = reinterpret_cast<KernelArgs*>(raw_arg);

  // Tile base addresses in shared memory (128 KiB total)
  const uint32_t smem_A = 0x00000000;
  const uint32_t smem_B = 0x00004000; // 16 KiB in bytes

  // Total elements in A_tile and B_tile
  const uint32_t MN = args->M * args->N;
  const uint32_t MK = args->M * args->K;
  const uint32_t KN = args->K * args->N;

  // 1. Cooperatively load A (no transpose)
  for (uint32_t i = tid_in_threadblock; i < MK; i += threads_per_threadblock) {
    uint32_t gaddr = reinterpret_cast<uint32_t>(&args->A[i]);
    uint32_t val;
    asm volatile("lw.global %0, 0(%1)" : "=r"(val) : "r"(gaddr) : "memory");
    store_shared(smem_A, i * 4, val);
  }

  // 2. Cooperatively load B (transposed and padded)
  for (uint32_t i = tid_in_threadblock; i < KN; i += threads_per_threadblock) {
    uint32_t gaddr = reinterpret_cast<uint32_t>(&args->B[i]);
    uint32_t val;
    asm volatile("lw.global %0, 0(%1)" : "=r"(val) : "r"(gaddr) : "memory");

    // Transpose index: (k, n) -> (n, k)
    // B is KxN -> B^T is NxK
    uint32_t k = i / args->N;
    uint32_t n = i % args->N;
    
    // Store into padded layout: stride = K + PAD = 64 + 1 = 65
    uint32_t smem_idx = n * 65 + k;
    store_shared(smem_B, smem_idx * 4, val);
  }

  // 3. Synchronize across cores to ensure SMEM is fully populated
  mu_barrier(0, threads_per_threadblock / 16); // All 8 warps

  // 4. Compute using bank-conflict-free access pattern
  for (uint32_t idx = tid_in_threadblock; idx < MN; idx += threads_per_threadblock) {
    uint32_t row = idx / args->N;
    uint32_t col_base = idx % args->N;

    float acc0 = 0.0f;
    float acc1 = 0.0f;
    float acc2 = 0.0f;
    float acc3 = 0.0f;

    // Pointers into SMEM tiles
    uint32_t pA = smem_A + row * args->K * 4;
    uint32_t pB0 = smem_B + (col_base + 0) * 65 * 4;
    uint32_t pB1 = smem_B + (col_base + 16) * 65 * 4;
    uint32_t pB2 = smem_B + (col_base + 32) * 65 * 4;
    uint32_t pB3 = smem_B + (col_base + 48) * 65 * 4;

    // Vectorized inner product with two-element unroll
    uint32_t k_end = args->K;
    for (uint32_t k = 0; k < k_end; k += 2) {
      float a0 = load32_shared_offset(pA, 0);
      float a1 = load32_shared_offset(pA, 4);

      acc0 += a0 * load32_shared_offset(pB0, 0);
      acc1 += a0 * load32_shared_offset(pB1, 0);
      acc2 += a0 * load32_shared_offset(pB2, 0);
      acc3 += a0 * load32_shared_offset(pB3, 0);

      acc0 += a1 * load32_shared_offset(pB0, 4);
      acc1 += a1 * load32_shared_offset(pB1, 4);
      acc2 += a1 * load32_shared_offset(pB2, 4);
      acc3 += a1 * load32_shared_offset(pB3, 4);

      pA += 8;
      pB0 += 8;
      pB1 += 8;
      pB2 += 8;
      pB3 += 8;
    }

    // Store results to global memory directly
    args->C[row * args->N + col_base + 0]  = acc0;
    args->C[row * args->N + col_base + 16] = acc1;
    args->C[row * args->N + col_base + 32] = acc2;
    args->C[row * args->N + col_base + 48] = acc3;
  }
}
