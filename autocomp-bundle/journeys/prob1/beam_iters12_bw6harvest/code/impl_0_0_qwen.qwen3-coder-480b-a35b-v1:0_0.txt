static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t CKK = a->C * a->KH * a->KW;
  const uint32_t smem_w = a->OH * a->OW * CKK * 4; // W staged AFTER patches

  // (1) stage W flat into SMEM[smem_w..].
  {
    const uint32_t total_w = a->OC * CKK;
    for (uint32_t i = tid_in_threadblock; i < total_w; i += threads_per_threadblock) {
      store_shared(smem_w + i * 4, 0, __builtin_bit_cast(uint32_t, a->w[i]));
    }
  }

  // (2) gather im2col patches into SMEM[0..] (eliminating div/mod)
  {
    const uint32_t KH = a->KH, KW = a->KW, OW = a->OW;
    const uint32_t Wd = a->W, HWd = a->H * Wd, stride = a->stride;

    if (__builtin_expect(KW == 16 && KH == 16 && OW == 4, 1)) {
      const uint32_t OH = a->OH;
      for (uint32_t ph = 0; ph < OH; ++ph) {
        const uint32_t ph_stride_Wd = ph * stride * Wd;
        for (uint32_t pw = 0; pw < 4; ++pw) {
          const uint32_t p = ph * 4 + pw;
          const uint32_t pw_stride = pw * stride;
          const uint32_t smem_p_base = p * CKK * 4;
          for (uint32_t ckk = tid_in_threadblock; ckk < CKK; ckk += threads_per_threadblock) {
            const uint32_t c = ckk >> 8;           // ckk / 256
            const uint32_t kh = (ckk >> 4) & 15;    // (ckk / 16) % 16
            const uint32_t kw = ckk & 15;           // ckk % 16
            const uint32_t in_idx = c * HWd + ph_stride_Wd + kh * Wd + pw_stride + kw;
            store_shared(smem_p_base + ckk * 4, 0, __builtin_bit_cast(uint32_t, a->in[in_idx]));
          }
        }
      }
    } else {
      // Generic fallback path for arbitrary shapes
      const uint32_t total = a->OH * OW * CKK;
      for (uint32_t i = tid_in_threadblock; i < total; i += threads_per_threadblock) {
        const uint32_t p = i / CKK;
        const uint32_t ckk = i % CKK;
        const uint32_t in_idx = (ckk / (KH * KW)) * HWd
                              + ((p / OW) * stride + (ckk / KW) % KH) * Wd
                              + (p % OW) * stride + ckk % KW;
        store_shared(i * 4, 0, __builtin_bit_cast(uint32_t, a->in[in_idx]));
      }
    }
  }

  // Cross-core SMEM staging barrier
  mu_barrier(0, threads_per_threadblock / MU_NUM_THREADS); // ID 0 = cross-core/cluster barrier

  // (3) frugal matmul: OUT[oc*P + p] = sum_ckk W[oc][ckk] * patch[p][ckk].
  {
    const uint32_t P = a->OH * a->OW;
    if (__builtin_expect(P == 16, 1)) {
      const uint32_t total_out = a->OC * 16;
      for (uint32_t idx = tid_in_threadblock; idx < total_out; idx += threads_per_threadblock) {
        uint32_t wa = smem_w + (idx >> 4) * CKK * 4; // W base (idx / 16)
        uint32_t pa = (idx & 15) * CKK * 4;          // patch base (idx % 16)
        float acc = 0.0f;
        for (uint32_t k = 0; k < CKK; k++) {
          acc += __builtin_bit_cast(float, load32_shared(wa)) *
                 __builtin_bit_cast(float, load32_shared(pa));
          wa += 4; pa += 4;
        }
        a->out[idx] = acc;
      }
    } else {
      // Generic fallback path for arbitrary shapes
      const uint32_t total_out = a->OC * P;
      for (uint32_t idx = tid_in_threadblock; idx < total_out; idx += threads_per_threadblock) {
        uint32_t wa = smem_w + (idx / P) * CKK * 4; // W base
        uint32_t pa = (idx % P) * CKK * 4;          // patch base
        float acc = 0.0f;
        for (uint32_t k = 0; k < CKK; k++) {
          acc += __builtin_bit_cast(float, load32_shared(wa)) *
                 __builtin_bit_cast(float, load32_shared(pa));
          wa += 4; pa += 4;
        }
        a->out[idx] = acc;
      }
    }
  }
}
