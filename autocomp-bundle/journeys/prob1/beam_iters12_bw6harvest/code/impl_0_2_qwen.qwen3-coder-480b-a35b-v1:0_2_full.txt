static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t CKK = a->C * a->KH * a->KW;
  const uint32_t smem_w = a->OH * a->OW * CKK * 4; // W staged AFTER patches

  // (1) stage W flat into SMEM[smem_w..] using 64-bit stores
  for (uint32_t i = tid_in_threadblock * 2; i < a->OC * CKK; i += threads_per_threadblock * 2) {
    uint32_t w0 = __builtin_bit_cast(uint32_t, a->w[i]);
    uint32_t w1 = __builtin_bit_cast(uint32_t, a->w[i + 1]);
    uint64_t packed_w = (uint64_t)w1 << 32 | w0;
    store64_shared(smem_w + i * 4, 0, packed_w);
  }

  // (2) gather im2col patches into SMEM[0..] using 64-bit stores
  {
    const uint32_t KH = a->KH, KW = a->KW, OW = a->OW;
    const uint32_t Wd = a->W, HWd = a->H * Wd, stride = a->stride;
    const uint32_t total = a->OH * OW * CKK;
    for (uint32_t i = tid_in_threadblock * 2; i < total; i += threads_per_threadblock * 2) {
      // First element
      const uint32_t p0 = i / CKK;
      const uint32_t ckk0 = i % CKK;
      const uint32_t in_idx0 = (ckk0 / (KH * KW)) * HWd
                             + ((p0 / OW) * stride + (ckk0 / KW) % KH) * Wd
                             + (p0 % OW) * stride + ckk0 % KW;
      uint32_t in0 = __builtin_bit_cast(uint32_t, a->in[in_idx0]);

      // Second element
      const uint32_t p1 = (i + 1) / CKK;
      const uint32_t ckk1 = (i + 1) % CKK;
      const uint32_t in_idx1 = (ckk1 / (KH * KW)) * HWd
                             + ((p1 / OW) * stride + (ckk1 / KW) % KH) * Wd
                             + (p1 % OW) * stride + ckk1 % KW;
      uint32_t in1 = __builtin_bit_cast(uint32_t, a->in[in_idx1]);

      uint64_t packed_in = (uint64_t)in1 << 32 | in0;
      store64_shared(i * 4, 0, packed_in);
    }
  }

  // Cross-core SMEM staging: synchronize across all 8 warps
  mu_barrier(0, threads_per_threadblock / MU_NUM_THREADS);

  // (3) frugal matmul: OUT[oc*P + p] = sum_ckk W[oc][ckk] * patch[p][ckk].
  {
    const uint32_t P = a->OH * a->OW;
    for (uint32_t idx = tid_in_threadblock; idx < a->OC * P; idx += threads_per_threadblock) {
      uint32_t wa = smem_w + (idx / P) * CKK * 4; // W base (after patches)
      uint32_t pa = (idx % P) * CKK * 4;          // patch base (SMEM offset 0)
      float acc = 0.0f;
      for (uint32_t k = 0; k < CKK; k++) {
        acc += __builtin_bit_cast(float, load32_shared(wa)) *
               __builtin_bit_cast(float, load32_shared(pa));
        wa += 4; pa += 4;
      }
      a->out[idx] = acc;
    }
  }
}
