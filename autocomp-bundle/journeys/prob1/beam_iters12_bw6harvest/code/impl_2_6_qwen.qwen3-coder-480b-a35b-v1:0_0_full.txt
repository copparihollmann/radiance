#include <stdint.h>

#define MU_NUM_THREADS 16

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t CKK = a->C * a->KH * a->KW;

  // Padding the inner dimension from 768 to 784 (next multiple of 16) to avoid bank conflicts
  const uint32_t CKK_PADDED = (CKK + 15) & ~15;
  const uint32_t smem_w = a->OH * a->OW * CKK_PADDED * 4; // W staged AFTER patches

  // (1) stage W flat into SMEM[smem_w..] with padding
  {
    const uint32_t total_w = a->OC * CKK;
    for (uint32_t i = tid_in_threadblock; i < total_w; i += threads_per_threadblock) {
      const uint32_t oc = i / CKK;
      const uint32_t ckk = i % CKK;
      const uint32_t smem_addr = smem_w + oc * CKK_PADDED * 4 + ckk * 4;
      store_shared(smem_addr, 0, __builtin_bit_cast(uint32_t, a->w[i]));
    }
  }

  // (2) gather im2col patches into SMEM[0..] with padding
  {
    const uint32_t KH = a->KH, KW = a->KW, OW = a->OW;
    const uint32_t Wd = a->W, HWd = a->H * Wd, stride = a->stride;

    const uint32_t total = a->OH * OW * CKK;
    for (uint32_t i = tid_in_threadblock; i < total; i += threads_per_threadblock) {
      const uint32_t p = i / CKK;
      const uint32_t ckk = i % CKK;
      const uint32_t ph = p / OW;
      const uint32_t pw = p % OW;
      const uint32_t c = ckk / (KH * KW);
      const uint32_t kh = (ckk / KW) % KH;
      const uint32_t kw = ckk % KW;
      const uint32_t in_idx = c * HWd + (ph * stride + kh) * Wd + (pw * stride + kw);
      const uint32_t smem_addr = p * CKK_PADDED * 4 + ckk * 4;
      store_shared(smem_addr, 0, __builtin_bit_cast(uint32_t, a->in[in_idx]));
    }
  }

  // Cross-core SMEM staging barrier
  mu_barrier(0, threads_per_threadblock / MU_NUM_THREADS);

  // (3) Matmul phase
  {
    const uint32_t P = a->OH * a->OW;
    const uint32_t total_out = a->OC * P;
    for (uint32_t idx = tid_in_threadblock; idx < total_out; idx += threads_per_threadblock) {
      const uint32_t oc = idx / P;
      const uint32_t p = idx % P;
      uint32_t wa = smem_w + oc * CKK_PADDED * 4;
      uint32_t pa = p * CKK_PADDED * 4;
      float acc = 0.0f;
      
      // Unrolled by 4 to balance performance and register usage
      uint32_t k = 0;
      for (; k + 3 < CKK; k += 4) {
        float w0 = __builtin_bit_cast(float, load32_shared(wa + 0));
        float p0 = __builtin_bit_cast(float, load32_shared(pa + 0));
        acc += w0 * p0;
        float w1 = __builtin_bit_cast(float, load32_shared(wa + 4));
        float p1 = __builtin_bit_cast(float, load32_shared(pa + 4));
        acc += w1 * p1;
        float w2 = __builtin_bit_cast(float, load32_shared(wa + 8));
        float p2 = __builtin_bit_cast(float, load32_shared(pa + 8));
        acc += w2 * p2;
        float w3 = __builtin_bit_cast(float, load32_shared(wa + 12));
        float p3 = __builtin_bit_cast(float, load32_shared(pa + 12));
        acc += w3 * p3;
        wa += 16;
        pa += 16;
      }
      for (; k < CKK; k++) {
        float w = __builtin_bit_cast(float, load32_shared(wa));
        float p = __builtin_bit_cast(float, load32_shared(pa));
        acc += w * p;
        wa += 4;
        pa += 4;
      }
      a->out[idx] = acc;
    }
  }
}
