#include <stdint.h>

#define MU_NUM_THREADS 16

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t CKK = a->C * a->KH * a->KW;
  const uint32_t smem_w = a->OH * a->OW * CKK * 4; // W staged AFTER patches

  // (1) Stage W flat into SMEM[smem_w..], fully coalesced
  {
    const uint32_t total_w = a->OC * CKK;
    for (uint32_t i = tid_in_threadblock; i < total_w; i += threads_per_threadblock) {
      store_shared(smem_w + i * 4, 0, __builtin_bit_cast(uint32_t, a->w[i]));
    }
  }

  // (2) Gather im2col patches into SMEM[0..]
  {
    const uint32_t KH = a->KH, KW = a->KW, OW = a->OW;
    const uint32_t Wd = a->W, HWd = a->H * Wd, stride = a->stride;

    if (__builtin_constant_p(KW) && __builtin_constant_p(KH) && __builtin_constant_p(OW) &&
        KW == 16 && KH == 16 && OW == 4) {
      
      const uint32_t stride_Wd = stride * Wd;
      
      // Loop restructuring: process CKK chunks first to minimize integer div/mod
      #pragma GCC unroll 1
      for (uint32_t s = 0; s < 6; ++s) {
        const uint32_t ckk = tid_in_threadblock + (s << 7); // s * 128
        const uint32_t c = ckk >> 8;
        const uint32_t kh = (ckk >> 4) & 15;
        const uint32_t kw = ckk & 15;
        const uint32_t c_kh_kw_offset = c * HWd + kh * Wd + kw;
        
        #pragma GCC unroll 1
        for (uint32_t p = 0; p < 16; ++p) {
          const uint32_t ph = p >> 2;
          const uint32_t pw = p & 3;
          const uint32_t in_idx = c_kh_kw_offset + ph * stride_Wd + pw * stride;
          store_shared((p * 768 + ckk) * 4, 0, __builtin_bit_cast(uint32_t, a->in[in_idx]));
        }
      }
    } else {
      // Generic fallback path for arbitrary shapes
      const uint32_t total = a->OH * OW * CKK;
      for (uint32_t i = tid_in_threadblock; i < total; i += threads_per_threadblock) {
        const uint32_t p = i / CKK;
        const uint32_t ckk = i % CKK;
        const uint32_t in_idx = (ckk / (KH * KW)) * HWd
                              + ((p / OW) * stride + (ckk / KW) % KH) * Wd
                              + (p % OW) * stride + ckk % KW;
        store_shared(i * 4, 0, __builtin_bit_cast(uint32_t, a->in[in_idx]));
      }
    }
  }

  // Cross-core SMEM staging barrier
  mu_barrier(0, threads_per_threadblock / MU_NUM_THREADS);

  // (3) Matmul phase
  {
    const uint32_t P = a->OH * a->OW;
    if (__builtin_constant_p(P) && P == 16) {
      // Map threadblock threads as 8 OC groups of 16 patches each
      const uint32_t t_oc = tid_in_threadblock / 16;
      const uint32_t t_p = tid_in_threadblock % 16;

      const uint32_t wa0_base = smem_w + t_oc * CKK * 4;
      const uint32_t wa1_base = smem_w + (t_oc + 8) * CKK * 4;
      const uint32_t pa_base = t_p * CKK * 4;

      float acc0 = 0.0f;
      float acc1 = 0.0f;

      uint32_t pa_addr = pa_base;
      uint32_t wa0_addr = wa0_base;
      uint32_t wa1_addr = wa1_base;

      // Loop kept rolled to stay within register limit
      #pragma GCC unroll 1
      for (uint32_t k = 0; k < CKK; ++k) {
        float p_val = __builtin_bit_cast(float, load32_shared(pa_addr));
        float w0_val = __builtin_bit_cast(float, load32_shared(wa0_addr));
        float w1_val = __builtin_bit_cast(float, load32_shared(wa1_addr));

        acc0 += w0_val * p_val;
        acc1 += w1_val * p_val;

        pa_addr += 4;
        wa0_addr += 4;
        wa1_addr += 4;
      }

      a->out[t_oc * 16 + t_p] = acc0;
      a->out[(t_oc + 8) * 16 + t_p] = acc1;
    } else {
      // Generic fallback path for arbitrary shapes
      const uint32_t total_out = a->OC * P;
      for (uint32_t idx = tid_in_threadblock; idx < total_out; idx += threads_per_threadblock) {
        uint32_t wa = smem_w + (idx / P) * CKK * 4;
        uint32_t pa = (idx % P) * CKK * 4;
        float acc = 0.0f;
        for (uint32_t k = 0; k < CKK; ++k) {
          acc += __builtin_bit_cast(float, load32_shared(wa)) *
                 __builtin_bit_cast(float, load32_shared(pa));
          wa += 4; pa += 4;
        }
        a->out[idx] = acc;
      }
    }
  }
}
