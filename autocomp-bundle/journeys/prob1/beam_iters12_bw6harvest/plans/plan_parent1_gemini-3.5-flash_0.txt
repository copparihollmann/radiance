An analysis of the baseline code reveals a major hardware-level bottleneck: **severe shared memory bank conflicts**.

### Inefficiency: 16-Way Shared Memory Bank Conflicts
In the original implementation:
* The input patches are staged in shared memory with a stride of `CKK = 768` floats.
* During the matrix multiplication step, threads in a warp access different patches $p$ at the same element index $k$. Lane $i$ accesses the address `(i * 768 + k) * 4`.
* Because `768` is a multiple of the bank count (16 or 32), `(i * 768 + k) % 16` yields the same value for all $i$. This causes a **16-way bank conflict**, forcing the hardware to serialize the 16 lane reads over 16 cycles instead of performing them in a single cycle.

### Optimization Plan
To eliminate bank conflicts and optimize performance:
1. **Pad Shared Memory Allocations (Strategy 8):** Increase the row dimension stride from `CKK` to `CKK_padded = 769` (coprime to 16). 
   * This shifts the address for lane $i$ to `(i * 769 + k) * 4`.
   * Since $\gcd(769, 16) = 1$, the bank indices `(i + k) % 16` form a perfect permutation across the 16 lanes, completely eliminating all bank conflicts.
2. **Eliminate Arithmetic Divisions in Loops:** Replace flat 1D loops for staging with 2D loops utilizing pointer accumulation. This completely removes expensive division (`/`) and modulo (`%`) operations inside the staging loops, reducing instruction count and register pressure.
3. **Loop Unrolling (ILP):** Unroll the inner $K$-loop by 4 to expose independent floating-point operations and hide the 2-cycle latency of FP32 instructions on Muon's half-throughput PEs.

### Optimized Kernel Code

```cpp
#include <mu_intrinsics.h>
#include <mu_schedule.h>
#include <stdint.h>

// Assuming KernelArgs struct layout is defined as follows in the harness:
struct KernelArgs {
  __global float* in;
  __global float* w;
  __global float* out;
  uint32_t C;
  uint32_t H;
  uint32_t W;
  uint32_t KH;
  uint32_t KW;
  uint32_t OC;
  uint32_t OH;
  uint32_t OW;
  uint32_t stride;
};

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);

  // Vision Transformer Patch-Embedding Tile Dimensions
  constexpr uint32_t OC = 16;
  constexpr uint32_t CKK = 768;
  constexpr uint32_t CKK_padded = 769; // Pad to prevent 16-bank conflict (769 is coprime to 16)
  constexpr uint32_t P = 16;
  constexpr uint32_t OW = 4;
  constexpr uint32_t KH = 16;
  constexpr uint32_t KW = 16;
  constexpr uint32_t HWd = 64 * 64;
  constexpr uint32_t Wd = 64;
  constexpr uint32_t stride = 16;
  constexpr uint32_t smem_w = P * CKK_padded * 4; // W staged after patches with padding

  // (1) Stage W with padding into SMEM[smem_w..].
  // Uses pointer accumulation to avoid division/modulo or multiplication in the loop.
  uint32_t w_global_offset = 0;
  uint32_t w_shared_offset = smem_w;
  for (uint32_t oc = 0; oc < OC; ++oc) {
    for (uint32_t k = tid_in_threadblock; k < CKK; k += threads_per_threadblock) {
      store_shared(w_shared_offset + k * 4, 0, __builtin_bit_cast(uint32_t, a->w[w_global_offset + k]));
    }
    w_global_offset += CKK;
    w_shared_offset += CKK_padded * 4;
  }

  // (2) Restructured gather of im2col patches into SMEM[0..] with padding
  uint32_t p_shared_offset = 0;
  for (uint32_t p = 0; p < P; ++p) {
    const uint32_t p_div_OW = p / OW; // Compiler optimizes to: p >> 2
    const uint32_t p_mod_OW = p % OW; // Compiler optimizes to: p & 3
    const uint32_t h_base = p_div_OW * stride;
    const uint32_t w_base = p_mod_OW * stride;
    
    for (uint32_t ckk = tid_in_threadblock; ckk < CKK; ckk += threads_per_threadblock) {
      const uint32_t c = ckk / (KH * KW);       // Compiler optimizes to: ckk >> 8
      const uint32_t rem = ckk % (KH * KW);     // Compiler optimizes to: ckk & 255
      const uint32_t kh = rem / KW;             // Compiler optimizes to: rem >> 4
      const uint32_t kw = rem % KW;             // Compiler optimizes to: rem & 15
      
      const uint32_t in_idx = c * HWd + (h_base + kh) * Wd + (w_base + kw);
      store_shared(p_shared_offset + ckk * 4, 0, __builtin_bit_cast(uint32_t, a->in[in_idx]));
    }
    p_shared_offset += CKK_padded * 4;
  }

  // Synchronize across both cores (Cluster-wide barrier ID 0)
  mu_barrier(0, threads_per_threadblock / MU_NUM_THREADS);

  // (3) Frugal matmul with optimized division mapping, padded strides to eliminate bank conflicts
  for (uint32_t idx = tid_in_threadblock; idx < OC * P; idx += threads_per_threadblock) {
    const uint32_t oc = idx / P; // Compiler optimizes to: idx >> 4
    const uint32_t p = idx % P;  // Compiler optimizes to: idx & 15
    
    uint32_t wa = smem_w + oc * CKK_padded * 4;
    uint32_t pa = p * CKK_padded * 4;
    float acc = 0.0f;
    
    #pragma GCC unroll 4
    for (uint32_t k = 0; k < CKK; ++k) {
      acc += __builtin_bit_cast(float, load32_shared(wa)) *
             __builtin_bit_cast(float, load32_shared(pa));
      wa += 4;
      pa += 4;
    }
    a->out[idx] = acc;
  }
}
```
