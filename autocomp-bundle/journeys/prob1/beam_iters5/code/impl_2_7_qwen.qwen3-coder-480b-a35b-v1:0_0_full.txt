#define MU_NUM_THREADS 16
#define MU_NUM_CORES 2
#define CKK 768
#define P 16
#define PADDED_CKK 769
#define SMEM_WEIGHTS_OFFSET (PADDED_CKK * P * 4)  // After padded patches

struct KernelArgs {
  float* in;
  float* w;
  float* out;
  uint32_t OC, C, KH, KW, H, W, OH, OW, stride;
};

static inline void store_shared(uint32_t base, uint32_t offset, uint32_t data) {
    asm volatile("sw.shared %2, %1(%0)" :: "r"(base), "I"(offset), "r"(data) : "memory");
}

static inline uint32_t load32_shared(uint32_t address) {
    uint32_t data;
    asm volatile("lw.shared %0, %1(%2)" : "=r"(data) : "I"(0), "r"(address) : "memory");
    return data;
}

static inline void mu_barrier(unsigned barrier_id, unsigned num_warps) {
    asm volatile ("vx_bar %0, %1" :: "r"(barrier_id), "r"(num_warps) : "memory");
}
