#include <stdint.h>

struct KernelArgs {
  const __global float* A;
  const __global float* B;
  __global float* C;
  uint32_t M;
  uint32_t N;
  uint32_t K;
};

// Helper to store to shared memory using runtime address calculation
static inline void store32_shared_runtime(uint32_t addr, uint32_t data) {
  asm volatile("sw.shared %1, 0(%0)" :: "r"(addr), "r"(data) : "memory");
}

// Helper to load from shared memory
static inline uint32_t load32_shared_runtime(uint32_t addr) {
  uint32_t data;
  asm volatile("lw.shared %0, 0(%1)" : "=r"(data) : "r"(addr) : "memory");
  return data;
}

static inline float bits_to_float(uint32_t bits) {
  return __builtin_bit_cast(float, bits);
}

static inline uint32_t float_to_bits(float f) {
  return __builtin_bit_cast(uint32_t, f);
}

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* args = reinterpret_cast<KernelArgs*>(raw_arg);

  const uint32_t M = args->M;
  const uint32_t N = args->N;
  const uint32_t K = args->K;

  // Tile sizes
  const uint32_t TILE_M = 64;
  const uint32_t TILE_N = 64;
  const uint32_t TILE_K = 16;

  // Bounds checking for output tile
  if (M < TILE_M || N < TILE_N) return;

  // Assign output tile to this threadblock
  const uint32_t tb_m = (threadblock_id / (N / TILE_N)) * TILE_M;
  const uint32_t tb_n = (threadblock_id % (N / TILE_N)) * TILE_N;

  if (tb_m >= M) return;

  // Shared memory layout (128KB total)
  // Note: Padding avoids bank conflicts (add 16 floats per row)
  const uint32_t smem_A_base = 0;
  const uint32_t smem_B_base = smem_A_base + (TILE_M * (TILE_K + 16)) * sizeof(float);
  const uint32_t smem_C_base = smem_B_base + (TILE_K * (TILE_N + 16)) * sizeof(float);

  // Clear accumulator tile in shared memory
  for (uint32_t i = tid_in_threadblock; i < TILE_M * TILE_N; i += threads_per_threadblock) {
    store32_shared_runtime(smem_C_base + i * sizeof(float), float_to_bits(0.0f));
  }
  mu_barrier(0, threads_per_threadblock / 16); // Cross-core barrier

  // Loop over K tiles
  for (uint32_t k = 0; k < K; k += TILE_K) {
    // Stage A tile: TILE_M x TILE_K
    for (uint32_t i = tid_in_threadblock; i < TILE_M * TILE_K; i += threads_per_threadblock) {
      uint32_t local_r = i / TILE_K;
      uint32_t local_c = i % TILE_K;
      uint32_t global_r = tb_m + local_r;
      uint32_t global_c = k + local_c;

      float val = 0.0f;
      if (global_r < M && global_c < K) {
        val = args->A[global_r * K + global_c];
      }
      uint32_t padded_addr = smem_A_base + (local_r * (TILE_K + 16) + local_c) * sizeof(float);
      store32_shared_runtime(padded_addr, float_to_bits(val));
    }

    // Stage B tile: TILE_K x TILE_N
    for (uint32_t i = tid_in_threadblock; i < TILE_K * TILE_N; i += threads_per_threadblock) {
      uint32_t local_r = i / TILE_N;
      uint32_t local_c = i % TILE_N;
      uint32_t global_r = k + local_r;
      uint32_t global_c = tb_n + local_c;

      float val = 0.0f;
      if (global_r < K && global_c < N) {
        val = args->B[global_r * N + global_c];
      }
      uint32_t padded_addr = smem_B_base + (local_r * (TILE_N + 16) + local_c) * sizeof(float);
      store32_shared_runtime(padded_addr, float_to_bits(val));
    }

    mu_barrier(0, threads_per_threadblock / 16); // Wait for staging

    // Compute partial products and accumulate
    for (uint32_t op = 0; op < (TILE_M * TILE_N + threads_per_threadblock - 1) / threads_per_threadblock; ++op) {
      uint32_t linear_index = op * threads_per_threadblock + tid_in_threadblock;
      if (linear_index >= TILE_M * TILE_N) continue;

      uint32_t r = linear_index / TILE_N;
      uint32_t c = linear_index % TILE_N;

      float sum = 0.0f;
      uint32_t a_row_addr = smem_A_base + r * (TILE_K + 16) * sizeof(float);
      uint32_t b_col_addr = smem_B_base + c * sizeof(float);

      for (uint32_t t = 0; t < TILE_K; ++t) {
        float a_val = bits_to_float(load32_shared_runtime(a_row_addr));
        float b_val = bits_to_float(load32_shared_runtime(b_col_addr));
        sum += a_val * b_val;
        a_row_addr += sizeof(float);
        b_col_addr += (TILE_N + 16) * sizeof(float);
      }

      uint32_t c_addr = smem_C_base + linear_index * sizeof(float);
      float old_acc = bits_to_float(load32_shared_runtime(c_addr));
      store32_shared_runtime(c_addr, float_to_bits(old_acc + sum));
    }

    mu_barrier(0, threads_per_threadblock / 16); // Sync before next tile
  }

  // Write results back to global memory
  for (uint32_t i = tid_in_threadblock; i < TILE_M * TILE_N; i += threads_per_threadblock) {
    uint32_t r = i / TILE_N;
    uint32_t c = i % TILE_N;
    uint32_t global_r = tb_m + r;
    uint32_t global_c = tb_n + c;

    if (global_r < M && global_c < N) {
      uint32_t addr = smem_C_base + i * sizeof(float);
      args->C[global_r * N + global_c] = bits_to_float(load32_shared_runtime(addr));
    }
  }
}
