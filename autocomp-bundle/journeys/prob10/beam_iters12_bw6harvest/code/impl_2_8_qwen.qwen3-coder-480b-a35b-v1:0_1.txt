#include <stdint.h>

#define MU_NUM_THREADS 16

struct KernelArgs {
  float* A;
  float* B;
  float* C;
  uint32_t M;
  uint32_t N;
  uint32_t K;
};

static inline float bits_to_float(uint32_t b) { union { uint32_t u; float f; } c; c.u = b; return c.f; }
static inline uint32_t float_to_bits(float f) { union { uint32_t u; float f; } c; c.f = f; return c.u; }

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* args = reinterpret_cast<KernelArgs*>(raw_arg);
  
  const uint32_t M = args->M;
  const uint32_t N = args->N;
  const uint32_t K = args->K;
  
  const uint32_t tid = tid_in_threadblock;
  const uint32_t num_warps = threads_per_threadblock / MU_NUM_THREADS;

  // Shared memory layout
  const uint32_t smem_a = 0;               // 16x16 tile -> 1024 bytes
  const uint32_t smem_b = 1024;            // 16x64 tile -> 4096 bytes (padded for bank conflict avoidance)
  
  // Thread mapping to 8x16 grid
  const uint32_t r_thread = tid >> 4;      // tid / 16
  const uint32_t c_thread = tid & 15;      // tid % 16
  
  // Loop over blocks of C
  for (uint32_t by = 0; by < (M + 7) / 8; by++) {
    const uint32_t base_y = by * 8 + r_thread;
    if (base_y >= M) continue;
    
    for (uint32_t bx = 0; bx < (N + 15) / 16; bx++) {
      const uint32_t base_x = bx * 16 + c_thread;
      
      float acc = 0.0f;
      
      for (uint32_t k = 0; k < K; k += 16) {
        // Stage A tile (8x16) - each thread loads 2 elements
        {
          const uint32_t a_row = base_y;
          const uint32_t a_col_base = k + c_thread;
          
          if (a_col_base < K) {
            store_shared(smem_a + (r_thread * 16 + c_thread) * 4, 0, 
                         float_to_bits(args->A[a_row * K + a_col_base]));
          }
          
          const uint32_t a_col2 = k + c_thread + 16;
          if (a_col2 < K) {
            store_shared(smem_a + (r_thread * 16 + c_thread + 16) * 4, 0, 
                         float_to_bits(args->A[a_row * K + a_col2]));
          }
        }
        
        // Stage B tile (16x16) - each thread loads 2 elements
        {
          const uint32_t b_row_base = k + r_thread;
          const uint32_t b_col = bx * 16 + c_thread;
          
          if (b_row_base < K) {
            store_shared(smem_b + (r_thread * 16 + c_thread) * 4, 0, 
                         float_to_bits(args->B[b_row_base * N + b_col]));
          }
          
          const uint32_t b_row2 = k + r_thread + 16;
          if (b_row2 < K) {
            store_shared(smem_b + (r_thread * 16 + c_thread + 256) * 4, 0, 
                         float_to_bits(args->B[b_row2 * N + b_col]));
          }
        }
        
        // Cross-core barrier
        mu_barrier(0, num_warps);
        
        // Compute partial dot product
        for (uint32_t kk = 0; kk < 16 && (k + kk) < K; kk++) {
          float a_val = bits_to_float(load32_shared(smem_a + (r_thread * 16 + kk) * 4));
          float b_val = bits_to_float(load32_shared(smem_b + (kk * 16 + c_thread) * 4));
          acc += a_val * b_val;
        }
        
        // Another cross-core barrier before next iteration
        mu_barrier(0, num_warps);
      }
      
      // Write back result
      if (base_x < N) {
        args->C[base_y * N + base_x] = acc;
      }
    }
  }
}
