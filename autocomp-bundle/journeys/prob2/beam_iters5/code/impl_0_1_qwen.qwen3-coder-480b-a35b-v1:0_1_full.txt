static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t seq = 64;
  const uint32_t d = 64;
  const float scale = 0.125f;

  // Shared memory layout (128 KiB total)
  __local float* s_Q = (__local float*)0x00000000;          // 0x00000 - 0x04000 (16 KiB)
  __local float* s_K = (__local float*)0x00004000;          // 0x04000 - 0x08000 (16 KiB)
  __local float* s_V = (__local float*)0x00008000;          // 0x08000 - 0x0C000 (16 KiB)
  __local float* s_S = (__local float*)0x0000C000;          // 0x0C000 - 0x10000 (16 KiB)

  // Phase 0: Load Q, K, V into shared memory
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    s_Q[idx] = a->Q[idx];
  }
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    s_K[idx] = a->K[idx];
  }
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    s_V[idx] = a->V[idx];
  }

  // Cross-core barrier after loading
  mu_barrier(0, threads_per_threadblock / 16);

  // Phase 1: Compute S = Q * K^T * scale
  for (uint32_t idx = tid_in_threadblock; idx < seq * seq; idx += threads_per_threadblock) {
    const uint32_t i = idx / seq;
    const uint32_t j = idx % seq;
    
    float acc = 0.0f;
#pragma GCC unroll 1
    for (uint32_t k = 0; k < d; k++) {
      acc += s_Q[i * d + k] * s_K[j * d + k];
    }
    s_S[idx] = acc * scale;
  }

  // Cross-core barrier after score computation
  mu_barrier(0, threads_per_threadblock / 16);

  // Phase 2: Softmax per row
  // Distribute rows: Core 0 handles rows 0..31, Core 1 handles rows 32..63
  uint32_t row = 0xFFFFFFFF; // Invalid row by default
  if (tid_in_threadblock < 32) {
    row = tid_in_threadblock;
  } else if (tid_in_threadblock >= 64 && tid_in_threadblock < 96) {
    row = tid_in_threadblock - 64 + 32;
  }

  if (row < seq) {
    float max_val = s_S[row * seq];
#pragma GCC unroll 1
    for (uint32_t j = 1; j < seq; j++) {
      float val = s_S[row * seq + j];
      if (val > max_val) max_val = val;
    }

    float sum = 0.0f;
#pragma GCC unroll 1
    for (uint32_t j = 0; j < seq; j++) {
      float e = mu_exp(s_S[row * seq + j] - max_val);
      s_S[row * seq + j] = e;
      sum += e;
    }

    const float inv_sum = 1.0f / sum;
#pragma GCC unroll 1
    for (uint32_t j = 0; j < seq; j++) {
      s_S[row * seq + j] *= inv_sum;
    }
  }

  // Cross-core barrier after softmax
  mu_barrier(0, threads_per_threadblock / 16);

  // Phase 3: Output = Softmax * V
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    const uint32_t i = idx / d;
    const uint32_t k = idx % d;

    float acc = 0.0f;
#pragma GCC unroll 1
    for (uint32_t j = 0; j < seq; j++) {
      acc += s_S[i * seq + j] * s_V[j * d + k];
    }
    a->O[idx] = acc;
  }
}
