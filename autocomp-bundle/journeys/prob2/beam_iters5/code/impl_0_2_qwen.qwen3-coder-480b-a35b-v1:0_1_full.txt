#define MU_NUM_THREADS 16
#define Q_OFFSET 0
#define K_OFFSET 16640
#define V_OFFSET 33280
#define S_OFFSET 49920
#define ROW_STRIDE 65
#define PADDED_ROW_BYTES 260

static inline float load_float_shared(uint32_t addr) {
    uint32_t u = load32_shared(addr);
    return __builtin_bit_cast(float, u);
}

static inline void store_float_shared(uint32_t addr, float f) {
    uint32_t u = __builtin_bit_cast(uint32_t, f);
    store_shared(addr, 0, u);
}

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {

  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const float scale = 0.125f;
  const uint32_t num_warps = threads_per_threadblock / MU_NUM_THREADS;

  // Stage Q matrix
  for (uint32_t idx = tid_in_threadblock; idx < 4096; idx += threads_per_threadblock) {
    uint32_t r = idx >> 6;
    uint32_t c = idx & 63;
    uint32_t smem_addr = Q_OFFSET + (r * ROW_STRIDE + c) * 4;
    store_float_shared(smem_addr, a->Q[idx]);
  }

  // Stage K matrix
  for (uint32_t idx = tid_in_threadblock; idx < 4096; idx += threads_per_threadblock) {
    uint32_t r = idx >> 6;
    uint32_t c = idx & 63;
    uint32_t smem_addr = K_OFFSET + (r * ROW_STRIDE + c) * 4;
    store_float_shared(smem_addr, a->K[idx]);
  }

  // Stage V matrix
  for (uint32_t idx = tid_in_threadblock; idx < 4096; idx += threads_per_threadblock) {
    uint32_t r = idx >> 6;
    uint32_t c = idx & 63;
    uint32_t smem_addr = V_OFFSET + (r * ROW_STRIDE + c) * 4;
    store_float_shared(smem_addr, a->V[idx]);
  }

  // Cross-core synchronization after staging
  mu_barrier(0, num_warps);

  // Compute attention scores S = Q * K^T
  for (uint32_t idx = tid_in_threadblock; idx < 4096; idx += threads_per_threadblock) {
    uint32_t i = idx >> 6;
    uint32_t j = idx & 63;
    float acc = 0.0f;
    
    for (uint32_t k = 0; k < 64; k++) {
      float q = load_float_shared(Q_OFFSET + (i * ROW_STRIDE + k) * 4);
      float k_val = load_float_shared(K_OFFSET + (j * ROW_STRIDE + k) * 4);
      acc += q * k_val;
    }
    
    uint32_t s_addr = S_OFFSET + (i * ROW_STRIDE + j) * 4;
    store_float_shared(s_addr, acc * scale);
  }

  // Synchronize before softmax
  mu_barrier(0, num_warps);

  // Softmax computation
  if (tid_in_threadblock < 64) {
    uint32_t row = tid_in_threadblock;
    uint32_t row_base = S_OFFSET + row * ROW_STRIDE * 4;
    
    // Find row maximum
    float max_val = load_float_shared(row_base);
    for (uint32_t j = 1; j < 64; j++) {
      float val = load_float_shared(row_base + j * 4);
      if (val > max_val) max_val = val;
    }
    
    // Compute exponentials and sum
    float sum = 0.0f;
    for (uint32_t j = 0; j < 64; j++) {
      float val = load_float_shared(row_base + j * 4);
      float exp_val = mu_exp(val - max_val);
      store_float_shared(row_base + j * 4, exp_val);
      sum += exp_val;
    }
    
    // Normalize
    float inv_sum = 1.0f / sum;
    for (uint32_t j = 0; j < 64; j++) {
      float val = load_float_shared(row_base + j * 4);
      store_float_shared(row_base + j * 4, val * inv_sum);
    }
  }

  // Synchronize after softmax
  mu_barrier(0, num_warps);

  // Compute output O = P * V
  for (uint32_t idx = tid_in_threadblock; idx < 4096; idx += threads_per_threadblock) {
    uint32_t i = idx >> 6;
    uint32_t k = idx & 63;
    float acc = 0.0f;
    
    for (uint32_t j = 0; j < 64; j++) {
      float p = load_float_shared(S_OFFSET + (i * ROW_STRIDE + j) * 4);
      float v = load_float_shared(V_OFFSET + (j * ROW_STRIDE + k) * 4);
      acc += p * v;
    }
    
    a->O[idx] = acc;
  }
}
