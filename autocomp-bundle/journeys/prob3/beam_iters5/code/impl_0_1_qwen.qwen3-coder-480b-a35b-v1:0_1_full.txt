static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t seq = a->seq;
  const uint32_t d = a->d;
  const uint32_t num_warps = threads_per_threadblock / MU_NUM_THREADS;
  const float scale = 0.125f;

  // Phase 1: scores (unchanged)
  for (uint32_t idx = tid_in_threadblock; idx < seq * seq; idx += threads_per_threadblock) {
    const uint32_t i = idx / seq;
    const uint32_t j = idx % seq;
    float acc = 0.0f;
    for (uint32_t k = 0; k < d; k++) {
      acc += a->Q[i * d + k] * a->K[j * d + k];
    }
    a->scratch[idx] = acc * scale;
  }
  mu_barrier(1, num_warps);

  // Phase 2: Optimized Softmax using shared memory reductions
  const uint32_t shared_offset_max = 0;                       // Offset for max values
  const uint32_t shared_offset_exp = shared_offset_max + 128 * sizeof(float); // Offset for exp values
  float* shared_max = (float*)shared_offset_max;
  float* shared_exp = (float*)shared_offset_exp;

  for (uint32_t row = 0; row < seq; row++) {
    // Cooperative load into shared memory
    float val = -1e30f; // Default for inactive threads
    if (tid_in_threadblock < seq) {
      val = a->scratch[row * seq + tid_in_threadblock];
    }
    shared_max[tid_in_threadblock] = val;
    mu_barrier(0, num_warps); // Ensure visibility across cores

    // Hierarchical reduction for max in shared memory
    for (uint32_t stride = 64; stride > 0; stride >>= 1) {
      if (tid_in_threadblock < stride) {
        float lhs = shared_max[tid_in_threadblock];
        float rhs = shared_max[tid_in_threadblock + stride];
        shared_max[tid_in_threadblock] = (lhs > rhs) ? lhs : rhs;
      }
      mu_barrier(0, num_warps);
    }

    // Broadcast max to all threads
    float m = shared_max[0];

    // Compute exponentials and store in shared memory
    val = 0.0f; // Default for inactive threads
    if (tid_in_threadblock < seq) {
      val = mu_exp(a->scratch[row * seq + tid_in_threadblock] - m);
    }
    shared_exp[tid_in_threadblock] = val;
    mu_barrier(0, num_warps); // Ensure visibility across cores

    // Hierarchical reduction for sum in shared memory
    for (uint32_t stride = 64; stride > 0; stride >>= 1) {
      if (tid_in_threadblock < stride) {
        float lhs = shared_exp[tid_in_threadblock];
        float rhs = shared_exp[tid_in_threadblock + stride];
        shared_exp[tid_in_threadblock] = lhs + rhs;
      }
      mu_barrier(0, num_warps);
    }

    // Broadcast inverse sum to all threads
    float inv_sum = 1.0f / shared_exp[0];

    // Normalize and write back to global memory
    if (tid_in_threadblock < seq) {
      a->scratch[row * seq + tid_in_threadblock] =
          shared_exp[tid_in_threadblock] * inv_sum;
    }
    mu_barrier(0, num_warps); // Ensure completion before next iteration
  }

  // Phase 3: O = P V (unchanged)
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    const uint32_t i = idx / d;
    const uint32_t k = idx % d;
    float acc = 0.0f;
    for (uint32_t j = 0; j < seq; j++) {
      acc += a->scratch[i * seq + j] * a->V[j * d + k];
    }
    a->O[idx] = acc;
  }
}
