```cpp
#include <vx_intrinsics.h>
#include <mu_intrinsics.h>
#include <mu_schedule.h>
#include <shared_mem.h>
#include <stdint.h>

struct KernelArgs {
  __global float* Q;
  __global float* K;
  __global float* V;
  __global float* O;
  __global float* scratch;
  uint32_t seq;
  uint32_t d;
};

// Shared memory base pointer (128 KiB total)
__shared float* const s_scratch = reinterpret_cast<__shared float*>(0x0);

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;

  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t seq = a->seq;
  const uint32_t d = a->d;
  const uint32_t num_warps = threads_per_threadblock / MU_NUM_THREADS;
  const float scale = 0.125f; // 1/sqrt(64)

  // Phase 1: Compute scaled dot products and write to shared memory
  for (uint32_t idx = tid_in_threadblock; idx < seq * seq; idx += threads_per_threadblock) {
    const uint32_t i = idx / seq;
    const uint32_t j = idx % seq;
    float acc = 0.0f;
    for (uint32_t k = 0; k < d; k++) {
      acc += a->Q[i * d + k] * a->K[j * d + k];
    }
    s_scratch[idx] = acc * scale;
  }

  // Cross-core barrier to ensure all scores are written to SMEM before any thread reads them
  mu_barrier(0, num_warps);

  // Phase 2: Softmax row-wise in shared memory
  for (uint32_t row = tid_in_threadblock; row < seq; row += threads_per_threadblock) {
    float max_val = s_scratch[row * seq + 0];
    for (uint32_t j = 1; j < seq; j++) {
      float val = s_scratch[row * seq + j];
      if (val > max_val) {
        max_val = val;
      }
    }

    float sum = 0.0f;
    for (uint32_t j = 0; j < seq; j++) {
      float diff = s_scratch[row * seq + j] - max_val;
      float exp_val = mu_exp(diff);
      s_scratch[row * seq + j] = exp_val;
      sum += exp_val;
    }

    float inv_sum = 1.0f / sum;
    for (uint32_t j = 0; j < seq; j++) {
      s_scratch[row * seq + j] *= inv_sum;
    }
  }

  // Cross-core barrier after softmax to make updated values visible to all threads
  mu_barrier(0, num_warps);

  // Phase 3: Multiply softmax result (P) with V matrix
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    const uint32_t i = idx / d;
    const uint32_t k = idx % d;
    float acc = 0.0f;
    for (uint32_t j = 0; j < seq; j++) {
      acc += s_scratch[i * seq + j] * a->V[j * d + k];
    }
    a->O[idx] = acc;
  }
}
```
