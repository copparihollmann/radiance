### Selected Optimization Strategy: **Unified Working-Set In-Memory Colocation (FPMC)**

#### Explanation
Instead of implementing a complex, register-heavy tiling or FlashAttention-style streaming mechanism, we observe that for $SEQ=96$ and $HEAD\_DIM(d)=64$, the entire problem's working set—including all inputs ($Q$, $K$, $V$) and the intermediate attention scores ($S$)—can fit simultaneously inside the cluster's **128 KiB of Shared Memory (SMEM)**:
- $Q$: $96 \times 64 \times 4 \text{ B} = 24 \text{ KiB}$
- $K$: $96 \times 64 \times 4 \text{ B} = 24 \text{ KiB}$
- $V$: $96 \times 64 \times 4 \text{ B} = 24 \text{ KiB}$
- $S$ (Scratch): $96 \times 96 \times 4 \text{ B} = 36 \text{ KiB}$

**Total SMEM Required:** $108 \text{ KiB} \le 128 \text{ KiB}$.

By loading the entirety of $Q$, $K$, and $V$ into SMEM once at the beginning of the kernel cooperatively using all 128 threads, we can perform all subsequent phases (Scores, Softmax, and Output Projection) entirely within ultra-fast SMEM. This completely eliminates any DRAM reads or writes during intermediate phases. Furthermore, because we do not need to manage complex tiling loops, we can keep the loop structures simple and rolled (enforced via `#pragma GCC unroll 1`). This guarantees that we stay well below the strict hardware register limit of **31 distinct registers**, avoiding RTL-level execution failures while maximizing performance.

### Optimized Code

```cpp
#define SMEM_Q_OFFSET 0
#define SMEM_K_OFFSET 24576
#define SMEM_V_OFFSET 49152
#define SMEM_S_OFFSET 73728

inline void store_shared_float(uint32_t addr, float val) {
    union { float f; uint32_t i; } u;
    u.f = val;
    asm volatile("sw.shared %2, 0(%0)" :: "r"(addr), "r"(u.i) : "memory");
}

inline float load_shared_float(uint32_t addr) {
    union { float f; uint32_t i; } u;
    asm volatile("lw.shared %0, 0(%1)" : "=r"(u.i) : "r"(addr) : "memory");
    return u.f;
}

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t num_warps = threads_per_threadblock / MU_NUM_THREADS;
  const float scale = 0.125f; // 1/sqrt(64)

  // 1. Cooperative Global-to-SMEM Load of Q, K, and V
  const uint32_t total_elements = 96 * 64;
  
  #pragma GCC unroll 1
  for (uint32_t idx = tid_in_threadblock; idx < total_elements; idx += threads_per_threadblock) {
    store_shared_float(SMEM_Q_OFFSET + idx * 4, a->Q[idx]);
  }
  
  #pragma GCC unroll 1
  for (uint32_t idx = tid_in_threadblock; idx < total_elements; idx += threads_per_threadblock) {
    store_shared_float(SMEM_K_OFFSET + idx * 4, a->K[idx]);
  }
  
  #pragma GCC unroll 1
  for (uint32_t idx = tid_in_threadblock; idx < total_elements; idx += threads_per_threadblock) {
    store_shared_float(SMEM_V_OFFSET + idx * 4, a->V[idx]);
  }
  
  // Cross-Core Barrier to ensure all loaded values are visible to both cores
  mu_barrier(0, num_warps);

  // 2. Phase 1: Scores computation (completely inside SMEM)
  #pragma GCC unroll 1
  for (uint32_t idx = tid_in_threadblock; idx < 96 * 96; idx += threads_per_threadblock) {
    const uint32_t i = idx / 96;
    const uint32_t j = idx % 96;
    float acc = 0.0f;
    uint32_t q_addr = SMEM_Q_OFFSET + (i * 64) * 4;
    uint32_t k_addr = SMEM_K_OFFSET + (j * 64) * 4;
    
    #pragma GCC unroll 1
    for (uint32_t k = 0; k < 64; k++) {
      float q_val = load_shared_float(q_addr + k * 4);
      float k_val = load_shared_float(k_addr + k * 4);
      acc += q_val * k_val;
    }
    store_shared_float(SMEM_S_OFFSET + idx * 4, acc * scale);
  }
  
  mu_barrier(0, num_warps);

  // 3. Phase 2: Softmax per row (completely inside SMEM)
  #pragma GCC unroll 1
  for (uint32_t row = tid_in_threadblock; row < 96; row += threads_per_threadblock) {
    uint32_t row_addr = SMEM_S_OFFSET + (row * 96) * 4;
    float m = load_shared_float(row_addr);
    
    #pragma GCC unroll 1
    for (uint32_t j = 1; j < 96; j++) {
      const float v = load_shared_float(row_addr + j * 4);
      if (v > m) m = v;
    }
    
    float sum = 0.0f;
    #pragma GCC unroll 1
    for (uint32_t j = 0; j < 96; j++) {
      const float e = mu_exp(load_shared_float(row_addr + j * 4) - m);
      store_shared_float(row_addr + j * 4, e);
      sum += e;
    }
    
    const float inv = 1.0f / sum;
    #pragma GCC unroll 1
    for (uint32_t j = 0; j < 96; j++) {
      float val = load_shared_float(row_addr + j * 4);
      store_shared_float(row_addr + j * 4, val * inv);
    }
  }
  
  mu_barrier(0, num_warps);

  // 4. Phase 3: O = P V (reads S & V from SMEM, writes O to Global)
  #pragma GCC unroll 1
  for (uint32_t idx = tid_in_threadblock; idx < 96 * 64; idx += threads_per_threadblock) {
    const uint32_t i = idx / 64;
    const uint32_t k = idx % 64;
    float acc = 0.0f;
    uint32_t s_addr = SMEM_S_OFFSET + (i * 96) * 4;
    uint32_t v_addr = SMEM_V_OFFSET + k * 4;
    
    #pragma GCC unroll 1
    for (uint32_t j = 0; j < 96; j++) {
      float s_val = load_shared_float(s_addr + j * 4);
      float v_val = load_shared_float(v_addr + (j * 64) * 4);
      acc += s_val * v_val;
    }
    a->O[idx] = acc;
  }
}
```
