An elegant and highly optimized implementation of the FP32 single-head attention kernel for a sequence length of 96 and a head dimension of 64.

### Optimization Plan

1. **Shared Memory (SMEM) Staging**: We stage $Q$, $K$, and $V$ entirely into SMEM to eliminate redundant global memory loads. With $SEQ=96$ and $d=64$, the total memory footprint for $Q, K, V,$ and the intermediate scratchpad $S$ is 108 KiB, comfortably fitting within the 128 KiB SMEM limit.
2. **Column-Major Transposition of $V$ (Strategy 10)**: We transpose $V$ into a column-major layout during the SMEM loading phase. In the output GEMM phase ($O = S V$), this allows both $S$ and $V_{transposed}$ to be accessed with stride-1 (contiguous), maximizing cache lines and preventing SMEM bank conflicts.
3. **Register-Saving Address Calculations**: To adhere to the strict physical hardware constraint of $\le 31$ distinct registers, we avoid complex multi-dimensional index calculations inside the inner loops. Instead, we compute base addresses once and perform simple address increments (`ptr += 8`) within the loops. 
4. **Shallow Loop Unrolling**: We unroll the inner loops of the matrix multiplications and softmax calculations by a factor of 2. This exposes instruction-level parallelism (ILP) to hide the 2-cycle latency of the FP32 execution units, while avoiding register blowup from deep unrolling.
5. **Cross-Core Visibility Barrier (Rule 10)**: We use `mu_barrier(0, num_warps)` (ID 0) for all synchronizations to ensure that memory writes are fully coherent across both cores of the threadblock, preventing timing-dependent silent failures.

### Optimized C++ Kernel

```cpp
#include <stdint.h>

// Helper functions for loading/storing floats from/to SMEM using unions to avoid type-punning issues
static inline float load_float_shared(uint32_t byte_addr) {
    union { uint32_t i; float f; } u;
    u.i = load32_shared(byte_addr);
    return u.f;
}

static inline void store_float_shared(uint32_t byte_addr, float val) {
    union { float f; uint32_t i; } u;
    u.f = val;
    store_shared(byte_addr, 0, u.i);
}

static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t num_warps = threads_per_threadblock / MU_NUM_THREADS;
  const float scale = 0.125f; // 1/sqrt(64)

  // SMEM Offsets (Total size: 110,592 bytes < 128 KiB limit)
  const uint32_t Q_offset = 0;
  const uint32_t K_offset = 24576;
  const uint32_t V_offset = 49152;
  const uint32_t S_offset = 73728;

  // 1. Stage Q from global memory to SMEM
  for (uint32_t idx = tid_in_threadblock; idx < 6144; idx += 128) {
    float val = a->Q[idx];
    store_float_shared(Q_offset + idx * 4, val);
  }

  // 2. Stage K from global memory to SMEM
  for (uint32_t idx = tid_in_threadblock; idx < 6144; idx += 128) {
    float val = a->K[idx];
    store_float_shared(K_offset + idx * 4, val);
  }

  // 3. Stage V from global memory and transpose it into SMEM (column-major)
  // Original V is 96x64; transposed V is stored as 64x96 in SMEM
  for (uint32_t idx = tid_in_threadblock; idx < 6144; idx += 128) {
    const uint32_t j = idx / 64; // row in original V
    const uint32_t k = idx % 64; // col in original V
    float val = a->V[idx];
    store_float_shared(V_offset + (k * 96 + j) * 4, val);
  }

  // Synchronize to ensure Q, K, and V are fully staged across both cores
  mu_barrier(0, num_warps);

  // 4. Compute attention scores S = Q K^T * scale
  for (uint32_t idx = tid_in_threadblock; idx < 9216; idx += 128) {
    const uint32_t i = idx / 96;
    const uint32_t j = idx % 96;
    
    uint32_t q_addr = Q_offset + i * 256; // i * 64 * 4
    uint32_t k_addr = K_offset + j * 256; // j * 64 * 4
    float acc = 0.0f;
    
    // Unrolled by 2 to hide FP32 latency while minimizing register usage
    for (uint32_t k = 0; k < 64; k += 2) {
      float q_val0 = load_float_shared(q_addr);
      float k_val0 = load_float_shared(k_addr);
      float q_val1 = load_float_shared(q_addr + 4);
      float k_val1 = load_float_shared(k_addr + 4);
      acc += q_val0 * k_val0;
      acc += q_val1 * k_val1;
      q_addr += 8;
      k_addr += 8;
    }
    store_float_shared(S_offset + idx * 4, acc * scale);
  }

  // Synchronize to make scores visible for softmax
  mu_barrier(0, num_warps);

  // 5. Softmax per row in SMEM
  for (uint32_t row = tid_in_threadblock; row < 96; row += 128) {
    const uint32_t s_row_addr = S_offset + row * 384; // 96 * 4 = 384
    float m = -1e30f; // Initialize to a very small number
    
    // Find Max (unrolled by 2)
    uint32_t addr = s_row_addr;
    for (uint32_t j = 0; j < 96; j += 2) {
      float v0 = load_float_shared(addr);
      float v1 = load_float_shared(addr + 4);
      if (v0 > m) m = v0;
      if (v1 > m) m = v1;
      addr += 8;
    }
    
    // Compute Exp and Sum (unrolled by 2)
    float sum = 0.0f;
    addr = s_row_addr;
    for (uint32_t j = 0; j < 96; j += 2) {
      float v0 = load_float_shared(addr);
      float v1 = load_float_shared(addr + 4);
      float e0 = mu_exp(v0 - m);
      float e1 = mu_exp(v1 - m);
      store_float_shared(addr, e0);
      store_float_shared(addr + 4, e1);
      sum += e0 + e1;
      addr += 8;
    }
    
    // Scale by 1/sum (unrolled by 2)
    const float inv = 1.0f / sum;
    addr = s_row_addr;
    for (uint32_t j = 0; j < 96; j += 2) {
      float e0 = load_float_shared(addr);
      float e1 = load_float_shared(addr + 4);
      store_float_shared(addr, e0 * inv);
      store_float_shared(addr + 4, e1 * inv);
      addr += 8;
    }
  }

  // Synchronize to ensure softmax scaling is complete and visible
  mu_barrier(0, num_warps);

  // 6. Compute Output O = S V_transposed
  for (uint32_t idx = tid_in_threadblock; idx < 6144; idx += 128) {
    const uint32_t i = idx / 64;
    const uint32_t k = idx % 64;
    
    uint32_t s_addr = S_offset + i * 384; // i * 96 * 4
    uint32_t v_addr = V_offset + k * 384; // k * 96 * 4
    float acc = 0.0f;
    
    // Unrolled by 2 for stride-1 contiguous dot product in SMEM
    for (uint32_t j = 0; j < 96; j += 2) {
      float s_val0 = load_float_shared(s_addr);
      float v_val0 = load_float_shared(v_addr);
      float s_val1 = load_float_shared(s_addr + 4);
      float v_val1 = load_float_shared(v_addr + 4);
      acc += s_val0 * v_val0;
      acc += s_val1 * v_val1;
      s_addr += 8;
      v_addr += 8;
    }
    a->O[idx] = acc;
  }
}
```
