static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t seq = a->seq;
  const uint32_t d = a->d;
  const uint32_t nw = threads_per_threadblock / MU_NUM_THREADS;

  // Shared memory layout (all addresses are byte addresses)
  // s_K or s_V:      0x00000 to 0x07FFF (32 KiB = 8192 floats)
  // s_scratch:       0x08000 to 0x17FFF (64 KiB = 16384 floats)
  // s_Q or s_O:      0x18000 to 0x1FFFF (32 KiB = 8192 floats)
  constexpr uint32_t SMEM_K_OR_V_ADDR    = 0x00000; // 0x0000 to 0x7FFF
  constexpr uint32_t SMEM_SCRATCH_ADDR   = 0x08000; // 0x8000 to 0x17FFF
  constexpr uint32_t SMEM_Q_OR_O_ADDR    = 0x18000; // 0x18000 to 0x1FFFF

  // Cast shared memory addresses to float pointers for easier use
  float* s_K_or_V = reinterpret_cast<float*>(SMEM_K_OR_V_ADDR);
  float* s_scratch = reinterpret_cast<float*>(SMEM_SCRATCH_ADDR);
  float* s_Q_or_O = reinterpret_cast<float*>(SMEM_Q_OR_O_ADDR);

  // Assign global pointers with correct address space qualifier
  const __global float* g_Q = a->Q;
  const __global float* g_K = a->K;
  const __global float* g_V = a->V;
        __global float* g_O = a->O;

  // --- PHASE 1: GEMM Q * K^T -> scratch ---
  // Cooperatively load Q and K into shared memory
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    s_Q_or_O[idx] = g_Q[idx];
    s_K_or_V[idx] = g_K[idx];
  }
  mu_barrier(0, nw); // Cross-core barrier: K and Q are fully loaded

  // Compute Q * K^T (GEMM) in shared memory
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    for (uint32_t j = 0; j < seq; j++) {
      float acc = 0.0f;
      for (uint32_t k = 0; k < d; k++) {
        acc += s_Q_or_O[i * d + k] * s_K_or_V[j * d + k];
      }
      s_scratch[i * seq + j] = acc * 0.125f;
    }
  }
  mu_barrier(0, nw); // Cross-core barrier: s_scratch is fully written

  // --- PHASE 2: Softmax in-place on s_scratch ---
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    float* row_ptr = &s_scratch[i * seq];

    // Find max
    float m = row_ptr[0];
    for (uint32_t j = 1; j < seq; j++) {
      if (row_ptr[j] > m) m = row_ptr[j];
    }

    // Exponentiate and compute sum
    float sum = 0.0f;
    for (uint32_t j = 0; j < seq; j++) {
      const float e = mu_exp(row_ptr[j] - m);
      row_ptr[j] = e;
      sum += e;
    }

    // Normalize
    const float inv_sum = 1.0f / sum;
    for (uint32_t j = 0; j < seq; j++) {
      row_ptr[j] *= inv_sum;
    }
  }
  mu_barrier(0, nw); // Cross-core barrier: Softmax is complete

  // --- PHASE 3: GEMM scratch * V -> O ---
  // Cooperatively load V into shared memory (reusing s_K_or_V)
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    s_K_or_V[idx] = g_V[idx];
  }
  mu_barrier(0, nw); // Cross-core barrier: V is fully loaded

  // Compute scratch * V (GEMM) in shared memory
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    for (uint32_t k = 0; k < d; k++) {
      float acc = 0.0f;
      for (uint32_t j = 0; j < seq; j++) {
        acc += s_scratch[i * seq + j] * s_K_or_V[j * d + k];
      }
      s_Q_or_O[i * d + k] = acc;
    }
  }
  mu_barrier(0, nw); // Cross-core barrier: O is fully computed in s_Q_or_O

  // Cooperatively write O from shared memory to global memory
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    g_O[idx] = s_Q_or_O[idx];
  }
}
