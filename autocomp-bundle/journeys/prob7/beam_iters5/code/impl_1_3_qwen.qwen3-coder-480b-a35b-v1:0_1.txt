static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t seq = a->seq;
  const uint32_t d = a->d;
  const uint32_t nw = threads_per_threadblock / MU_NUM_THREADS;

  // Shared memory layout:
  // 0x00000 - 0x08000 (32KB): smem_Q
  // 0x08000 - 0x10000 (32KB): smem_K
  // 0x10000 - 0x20000 (64KB): smem_scratch
  // For Phase 3:
  // 0x00000 - 0x08000 (32KB): smem_V
  // 0x08000 - 0x10000 (32KB): smem_O

  const uint32_t smem_Q_base = 0x00000;
  const uint32_t smem_K_base = 0x08000;
  const uint32_t smem_scratch_base = 0x10000;
  const uint32_t smem_V_base = 0x00000;
  const uint32_t smem_O_base = 0x08000;

  // Phase 1: S[i][j] = scale * Q[i]·K[j]
  // Load Q and K into shared memory in a coalesced manner
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    const uint32_t i = idx / d;
    const uint32_t k = idx % d;
    const __global float* qi = a->Q + i * d;
    const __global float* ki = a->K + i * d;
    store_shared(smem_Q_base, idx * 4, __builtin_bit_cast(uint32_t, qi[k]));
    store_shared(smem_K_base, idx * 4, __builtin_bit_cast(uint32_t, ki[k]));
  }
  mu_barrier(0, nw); // Cross-core sync after SMEM writes

  // Compute attention scores in shared memory
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    for (uint32_t j = 0; j < seq; j++) {
      float acc = 0.0f;
      for (uint32_t k = 0; k < d; k++) {
        float q_val = __builtin_bit_cast(float, load32_shared(smem_Q_base + (i * d + k) * 4));
        float k_val = __builtin_bit_cast(float, load32_shared(smem_K_base + (j * d + k) * 4));
        acc += q_val * k_val;
      }
      store_shared(smem_scratch_base, (i * seq + j) * 4, __builtin_bit_cast(uint32_t, acc * 0.125f));
    }
  }
  mu_barrier(0, nw); // Cross-core sync after writing scores

  // Phase 2: Row-wise softmax in shared memory
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    // Find max
    float m = __builtin_bit_cast(float, load32_shared(smem_scratch_base + (i * seq) * 4));
    for (uint32_t j = 1; j < seq; j++) {
      float val = __builtin_bit_cast(float, load32_shared(smem_scratch_base + (i * seq + j) * 4));
      if (val > m) m = val;
    }

    // Compute exp and sum
    float sum = 0.0f;
    for (uint32_t j = 0; j < seq; j++) {
      float val = __builtin_bit_cast(float, load32_shared(smem_scratch_base + (i * seq + j) * 4));
      float e = mu_exp(val - m);
      store_shared(smem_scratch_base, (i * seq + j) * 4, __builtin_bit_cast(uint32_t, e));
      sum += e;
    }

    // Normalize
    float inv_sum = 1.0f / sum;
    for (uint32_t j = 0; j < seq; j++) {
      float e = __builtin_bit_cast(float, load32_shared(smem_scratch_base + (i * seq + j) * 4));
      store_shared(smem_scratch_base, (i * seq + j) * 4, __builtin_bit_cast(uint32_t, e * inv_sum));
    }
  }
  mu_barrier(0, nw); // Cross-core sync after softmax

  // Phase 3: O[i][k] = sum_j P[i][j] * V[j][k]
  // Load V into shared memory
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    const uint32_t j = idx / d;
    const uint32_t k = idx % d;
    const __global float* vj = a->V + j * d;
    store_shared(smem_V_base, idx * 4, __builtin_bit_cast(uint32_t, vj[k]));
  }
  mu_barrier(0, nw); // Cross-core sync after loading V

  // Compute output in shared memory
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    for (uint32_t k = 0; k < d; k++) {
      float acc = 0.0f;
      for (uint32_t j = 0; j < seq; j++) {
        float p_val = __builtin_bit_cast(float, load32_shared(smem_scratch_base + (i * seq + j) * 4));
        float v_val = __builtin_bit_cast(float, load32_shared(smem_V_base + (j * d + k) * 4));
        acc += p_val * v_val;
      }
      store_shared(smem_O_base, (i * d + k) * 4, __builtin_bit_cast(uint32_t, acc));
    }
  }
  mu_barrier(0, nw); // Cross-core sync after computing O

  // Write output from shared memory to global memory
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    const uint32_t i = idx / d;
    const uint32_t k = idx % d;
    float val = __builtin_bit_cast(float, load32_shared(smem_O_base + (i * d + k) * 4));
    a->O[i * d + k] = val;
  }
}
