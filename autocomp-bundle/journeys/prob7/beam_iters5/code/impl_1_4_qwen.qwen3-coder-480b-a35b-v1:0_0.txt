static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t seq = a->seq;
  const uint32_t d = a->d;
  const uint32_t nw = threads_per_threadblock / MU_NUM_THREADS;

  // Shared memory layout (128 KiB):
  // Buffer A (64 KiB): 0x00000 - 0x10000
  //   Phases 1-3: smem_Q (32 KiB) + smem_K (32 KiB)
  //   Phases 4-6: smem_V (32 KiB) + smem_O (32 KiB)
  // Buffer B (64 KiB): 0x10000 - 0x20000
  //   smem_S (64 KiB)

  constexpr uint32_t SMEM_BASE = 0x00000000;
  constexpr uint32_t SMEM_Q_OFFSET = 0x00000;     // 0x00000 - 0x08000 (32 KiB)
  constexpr uint32_t SMEM_K_OFFSET = 0x08000;     // 0x08000 - 0x10000 (32 KiB)
  constexpr uint32_t SMEM_V_OFFSET = 0x00000;     // 0x00000 - 0x08000 (32 KiB)
  constexpr uint32_t SMEM_O_OFFSET = 0x08000;     // 0x08000 - 0x10000 (32 KiB)
  constexpr uint32_t SMEM_S_OFFSET = 0x10000;     // 0x10000 - 0x20000 (64 KiB)

  // Phase 1: Cooperative loading of Q and K to SMEM
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    // Load Q
    const __global float* q_ptr = a->Q + idx;
    float q_val;
    asm volatile("lw.global %0, 0(%1)" : "=r"(q_val) : "r"(q_ptr) : "memory");
    store_shared(SMEM_BASE + SMEM_Q_OFFSET, idx * 4, *(uint32_t*)&q_val);

    // Load K
    const __global float* k_ptr = a->K + idx;
    float k_val;
    asm volatile("lw.global %0, 0(%1)" : "=r"(k_val) : "r"(k_ptr) : "memory");
    store_shared(SMEM_BASE + SMEM_K_OFFSET, idx * 4, *(uint32_t*)&k_val);
  }
  mu_barrier(0, nw); // Cross-core sync after loading Q and K

  // Phase 2: Score computation S[i][j] = scale * Q[i]·K[j] (compute in SMEM)
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    for (uint32_t j = 0; j < seq; j++) {
      float acc = 0.0f;
      for (uint32_t k = 0; k < d; k++) {
        float q_ik = *(float*)&load32_shared(SMEM_BASE + SMEM_Q_OFFSET + (i * d + k) * 4);
        float k_jk = *(float*)&load32_shared(SMEM_BASE + SMEM_K_OFFSET + (j * d + k) * 4);
        acc += q_ik * k_jk;
      }
      store_shared(SMEM_BASE + SMEM_S_OFFSET, (i * seq + j) * 4, *(uint32_t*)&(acc * 0.125f));
    }
  }
  mu_barrier(0, nw); // Cross-core sync after computing scores

  // Phase 3: Softmax in place on S (in SMEM)
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    // Find max
    float m = *(float*)&load32_shared(SMEM_BASE + SMEM_S_OFFSET + (i * seq) * 4);
    for (uint32_t j = 1; j < seq; j++) {
      float s_ij = *(float*)&load32_shared(SMEM_BASE + SMEM_S_OFFSET + (i * seq + j) * 4);
      if (s_ij > m) m = s_ij;
    }

    // Compute exp and sum
    float sum = 0.0f;
    for (uint32_t j = 0; j < seq; j++) {
      uint32_t s_addr = SMEM_BASE + SMEM_S_OFFSET + (i * seq + j) * 4;
      float s_ij = *(float*)&load32_shared(s_addr);
      float e = mu_exp(s_ij - m);
      store_shared(SMEM_BASE + SMEM_S_OFFSET, (i * seq + j) * 4, *(uint32_t*)&e);
      sum += e;
    }

    // Normalize
    float inv_sum = 1.0f / sum;
    for (uint32_t j = 0; j < seq; j++) {
      uint32_t s_addr = SMEM_BASE + SMEM_S_OFFSET + (i * seq + j) * 4;
      float e = *(float*)&load32_shared(s_addr);
      store_shared(SMEM_BASE + SMEM_S_OFFSET, (i * seq + j) * 4, *(uint32_t*)&(e * inv_sum));
    }
  }
  mu_barrier(0, nw); // Cross-core sync after softmax

  // Phase 4: Cooperative loading of V to SMEM (reuse Q's space)
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    const __global float* v_ptr = a->V + idx;
    float v_val;
    asm volatile("lw.global %0, 0(%1)" : "=r"(v_val) : "r"(v_ptr) : "memory");
    store_shared(SMEM_BASE + SMEM_V_OFFSET, idx * 4, *(uint32_t*)&v_val);
  }
  mu_barrier(0, nw); // Cross-core sync after loading V

  // Phase 5: Output computation O[i][k] = sum_j P[i][j] * V[j][k] (compute in SMEM)
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    for (uint32_t k = 0; k < d; k++) {
      float acc = 0.0f;
      for (uint32_t j = 0; j < seq; j++) {
        float p_ij = *(float*)&load32_shared(SMEM_BASE + SMEM_S_OFFSET + (i * seq + j) * 4);
        float v_jk = *(float*)&load32_shared(SMEM_BASE + SMEM_V_OFFSET + (j * d + k) * 4);
        acc += p_ij * v_jk;
      }
      store_shared(SMEM_BASE + SMEM_O_OFFSET, (i * d + k) * 4, *(uint32_t*)&acc);
    }
  }
  mu_barrier(0, nw); // Cross-core sync after computing O

  // Phase 6: Cooperative writing of O from SMEM to global memory
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    uint32_t o_val_bits = load32_shared(SMEM_BASE + SMEM_O_OFFSET + idx * 4);
    float o_val = *(float*)&o_val_bits;
    __global float* o_ptr = a->O + idx;
    asm volatile("sw.global %1, 0(%0)" :: "r"(o_ptr), "r"(*(uint32_t*)&o_val) : "memory");
  }
}
