static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t seq = a->seq;
  const uint32_t d = a->d;
  const uint32_t nw = threads_per_threadblock / MU_NUM_THREADS;

  // Shared memory layout:
  // Buffer A (64 KiB): 0x00000000 to 0x00010000
  //   smem_Q: 0x00000000, 32 KiB (128x64 floats)
  //   smem_K: 0x00008000, 32 KiB (128x64 floats)
  //   smem_V: 0x00000000, 32 KiB (128x64 floats)
  //   smem_O: 0x00008000, 32 KiB (128x64 floats)
  // Buffer B (64 KiB): 0x00010000 to 0x00020000
  //   smem_S: 0x00010000, 64 KiB (128x128 floats)
  __global float* smem_Q = (__global float*)0x00000000;
  __global float* smem_K = (__global float*)0x00008000;
  __global float* smem_S = (__global float*)0x00010000;
  __global float* smem_V = (__global float*)0x00000000;
  __global float* smem_O = (__global float*)0x00008000;

  // Phase 1: Cooperative load Q and K into SMEM
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    smem_Q[idx] = a->Q[idx];
    smem_K[idx] = a->K[idx];
  }
  mu_barrier(0, nw);

  // Phase 2: Compute scores S[i][j] = scale * Q[i]·K[j]
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    for (uint32_t j = 0; j < seq; j++) {
      float acc = 0.0f;
      for (uint32_t k = 0; k < d; k++) {
        acc += smem_Q[i * d + k] * smem_K[j * d + k];
      }
      smem_S[i * seq + j] = acc * 0.125f;
    }
  }
  mu_barrier(0, nw);

  // Phase 3: In-place softmax on S
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    float* sp = smem_S + i * seq;
    float m = sp[0];
    for (uint32_t j = 1; j < seq; j++) {
      if (sp[j] > m) m = sp[j];
    }
    float sum = 0.0f;
    for (uint32_t j = 0; j < seq; j++) {
      const float e = mu_exp(sp[j] - m);
      sp[j] = e;
      sum += e;
    }
    const float inv = 1.0f / sum;
    for (uint32_t j = 0; j < seq; j++) {
      sp[j] *= inv;
    }
  }
  mu_barrier(0, nw);

  // Phase 4: Cooperative load V into SMEM (reusing smem_Q space)
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    smem_V[idx] = a->V[idx];
  }
  mu_barrier(0, nw);

  // Phase 5: Compute output O[i][k] = sum_j P[i][j] * V[j][k]
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    for (uint32_t k = 0; k < d; k++) {
      float acc = 0.0f;
      for (uint32_t j = 0; j < seq; j++) {
        acc += smem_S[i * seq + j] * smem_V[j * d + k];
      }
      smem_O[i * d + k] = acc;
    }
  }
  mu_barrier(0, nw);

  // Phase 6: Write output O from SMEM to global memory
  for (uint32_t idx = tid_in_threadblock; idx < seq * d; idx += threads_per_threadblock) {
    a->O[idx] = smem_O[idx];
  }
}
