#include <stdint.h>

struct alignas(16) float4 {
    float x, y, z, w;
};

static inline void kernel_body(
    void* raw_arg,
    uint32_t tid_in_threadblock,
    uint32_t threads_per_threadblock,
    uint32_t threadblock_id
) {
    (void)threadblock_id;
    auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
    const uint32_t seq = a->seq;
    const uint32_t d = a->d;
    const uint32_t nw = threads_per_threadblock / MU_NUM_THREADS;

    const uint32_t tid = tid_in_threadblock;
    const uint32_t warp_id = tid / MU_NUM_THREADS;
    const uint32_t lane_id = tid % MU_NUM_THREADS;

    // Shared memory layout
    // Use shared memory to stage Q, K, V rows
    // We'll use smem_base for staging
    uint32_t smem_q_base = 0;
    uint32_t smem_k_base = smem_q_base + seq * d * sizeof(float);
    uint32_t smem_v_base = smem_k_base + seq * d * sizeof(float);

    // Phase 1: S[i][j] = scale * Q[i]·K[j]
    for (uint32_t i = tid; i < seq; i += threads_per_threadblock) {
        const __global float* qi = a->Q + i * d;
        __global float* si = a->scratch + i * seq;

        // Load Q[i] into shared memory
        for (uint32_t k = lane_id; k < d; k += MU_NUM_THREADS) {
            uint32_t addr = smem_q_base + (warp_id * d + k) * sizeof(float);
            asm volatile("sw.shared %1, 0(%0)" :: "r"(addr), "r"(qi[k]) : "memory");
        }

        mu_barrier(0, nw); // Cross-core barrier for shared memory visibility

        // Compute dot product with all K[j]
        for (uint32_t j = 0; j < seq; ++j) {
            const __global float* kj = a->K + j * d;
            float acc = 0.0f;

            // Vectorized dot product using float4
            const uint32_t vec_d = d / 4;
            const __global float4* q_vec = (__global float4*)(a->Q + i * d);
            const __global float4* k_vec = (__global float4*)kj;

            for (uint32_t v = lane_id; v < vec_d; v += MU_NUM_THREADS) {
                float4 qv = q_vec[v];
                float4 kv = k_vec[v];
                acc += qv.x * kv.x + qv.y * kv.y + qv.z * kv.z + qv.w * kv.w;
            }

            // Reduce across lanes
            for (int offset = MU_NUM_THREADS / 2; offset > 0; offset /= 2) {
                float temp;
                asm volatile("fsd %0, 0(sp)" :: "f"(acc));
                asm volatile("fld %0, 0(sp)" : "=f"(temp));
                temp = __builtin_shuffle(temp, offset);
                acc += temp;
            }

            if (lane_id == 0) {
                si[j] = acc * 0.125f;
            }
        }
    }

    mu_barrier(1, nw); // Per-core barrier

    // Phase 2: row softmax in place
    for (uint32_t i = tid; i < seq; i += threads_per_threadblock) {
        __global float* sp = a->scratch + i * seq;
        float m = sp[0];
        for (uint32_t j = 1; j < seq; j++) { if (sp[j] > m) m = sp[j]; }
        float sum = 0.0f;
        for (uint32_t j = 0; j < seq; j++) { const float e = mu_exp(sp[j] - m); sp[j] = e; sum += e; }
        const float inv = 1.0f / sum;
        for (uint32_t j = 0; j < seq; j++) sp[j] *= inv;
    }

    mu_barrier(1, nw); // Per-core barrier

    // Phase 3: O[i][k] = sum_j P[i][j] * V[j][k]
    for (uint32_t i = tid; i < seq; i += threads_per_threadblock) {
        __global float* oi = a->O + i * d;
        const __global float* si = a->scratch + i * seq;

        // Load P[i][:] into shared memory
        for (uint32_t j = lane_id; j < seq; j += MU_NUM_THREADS) {
            uint32_t addr = smem_k_base + (warp_id * seq + j) * sizeof(float);
            asm volatile("sw.shared %1, 0(%0)" :: "r"(addr), "r"(si[j]) : "memory");
        }

        mu_barrier(0, nw); // Cross-core barrier

        // Compute O[i][k] = sum_j P[i][j] * V[j][k]
        for (uint32_t k = 0; k < d; ++k) {
            float acc = 0.0f;

            for (uint32_t j = lane_id; j < seq; j += MU_NUM_THREADS) {
                uint32_t p_addr = smem_k_base + (warp_id * seq + j) * sizeof(float);
                float p_val;
                asm volatile("lw.shared %0, 0(%1)" : "=r"(p_val) : "r"(p_addr) : "memory");

                uint32_t v_addr = smem_v_base + (warp_id * d + k) * sizeof(float);
                float v_val;
                asm volatile("lw.shared %0, 0(%1)" : "=r"(v_val) : "r"(v_addr) : "memory");

                acc += p_val * v_val;
            }

            // Reduce across lanes
            for (int offset = MU_NUM_THREADS / 2; offset > 0; offset /= 2) {
                float temp;
                asm volatile("fsd %0, 0(sp)" :: "f"(acc));
                asm volatile("fld %0, 0(sp)" : "=f"(temp));
                temp = __builtin_shuffle(temp, offset);
                acc += temp;
            }

            if (lane_id == 0) {
                oi[k] = acc;
            }
        }
    }
}
