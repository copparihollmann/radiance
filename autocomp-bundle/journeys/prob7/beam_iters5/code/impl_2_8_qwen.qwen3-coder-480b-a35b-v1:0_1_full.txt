static inline void kernel_body(
  void* raw_arg,
  uint32_t tid_in_threadblock,
  uint32_t threads_per_threadblock,
  uint32_t threadblock_id
) {
  (void)threadblock_id;
  auto* a = reinterpret_cast<KernelArgs*>(raw_arg);
  const uint32_t seq = a->seq;
  const uint32_t d = a->d;
  const uint32_t nw = threads_per_threadblock / MU_NUM_THREADS;

  // Shared memory pointer for the attention matrix (128x128 floats = 64KB)
  __local float* smem_scratch = (__local float*)0x0;

  // Phase 1: Q.K^T -> shared memory
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    const __global float* qi = a->Q + i * d;
    for (uint32_t j = 0; j < seq; j++) {
      const __global float* kj = a->K + j * d;
      float acc = 0.0f;
      for (uint32_t k = 0; k < d; k++) {
        acc += qi[k] * kj[k];
      }
      smem_scratch[i * seq + j] = acc * 0.125f;
    }
  }
  // Cross-core barrier to ensure all writes to SMEM are visible
  mu_barrier(0, nw);

  // Phase 2: Softmax in shared memory
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    __local float* sp = smem_scratch + i * seq;
    float m = sp[0];
    for (uint32_t j = 1; j < seq; j++) {
      if (sp[j] > m) m = sp[j];
    }
    float sum = 0.0f;
    for (uint32_t j = 0; j < seq; j++) {
      const float e = mu_exp(sp[j] - m);
      sp[j] = e;
      sum += e;
    }
    const float inv = 1.0f / sum;
    for (uint32_t j = 0; j < seq; j++) {
      sp[j] *= inv;
    }
  }
  // Cross-core barrier after softmax update
  mu_barrier(0, nw);

  // Phase 3: P.V -> output
  for (uint32_t i = tid_in_threadblock; i < seq; i += threads_per_threadblock) {
    const __local float* sp = smem_scratch + i * seq;
    for (uint32_t k = 0; k < d; k++) {
      const __global float* vk = a->V + k;
      float acc = 0.0f;
      for (uint32_t j = 0; j < seq; j++) {
        acc += sp[j] * vk[j * d];
      }
      a->O[i * d + k] = acc;
    }
  }
}
